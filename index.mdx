export const VariantSwitcher = () => {
  const variants = [
    { key: "A", name: "Guided transfer" },
    { key: "B", name: "Control room" },
    { key: "C", name: "Trust first" },
  ]
  const readVariant = () => {
    if (typeof window === "undefined") return "A"
    const requested = new URLSearchParams(window.location.search).get("variant")?.toUpperCase()
    return variants.some((item) => item.key === requested) ? requested : "A"
  }
  const [variant, setVariant] = useState("A")

  useEffect(() => {
    setVariant(readVariant())
    const sync = () => setVariant(readVariant())
    window.addEventListener("popstate", sync)
    return () => window.removeEventListener("popstate", sync)
  }, [])

  const choose = (next) => {
    const url = new URL(window.location.href)
    url.searchParams.set("variant", next)
    window.history.replaceState({}, "", url)
    setVariant(next)
  }

  const cycle = (direction) => {
    const index = variants.findIndex((item) => item.key === variant)
    choose(variants[(index + direction + variants.length) % variants.length].key)
  }

  useEffect(() => {
    const onKeyDown = (event) => {
      const tag = event.target?.tagName
      if (["INPUT", "TEXTAREA"].includes(tag) || event.target?.isContentEditable) return
      if (event.key === "ArrowLeft") cycle(-1)
      if (event.key === "ArrowRight") cycle(1)
    }
    window.addEventListener("keydown", onKeyDown)
    return () => window.removeEventListener("keydown", onKeyDown)
  }, [variant])

  const current = variants.find((item) => item.key === variant)

  return (
    <div className="not-prose">
      {variant === "A" && <VariantA />}
      {variant === "B" && <VariantB />}
      {variant === "C" && <VariantC />}
      <div className="fixed bottom-5 left-1/2 z-50 flex -translate-x-1/2 items-center gap-1 rounded-full border border-white/15 bg-[#111016]/95 p-1.5 text-white shadow-2xl backdrop-blur">
        <button aria-label="Previous variant" onClick={() => cycle(-1)} className="h-9 w-9 rounded-full hover:bg-white/10">←</button>
        <div className="min-w-48 px-3 text-center text-sm font-medium">
          {current.key} — {current.name}
        </div>
        <button aria-label="Next variant" onClick={() => cycle(1)} className="h-9 w-9 rounded-full hover:bg-white/10">→</button>
      </div>
    </div>
  )
}
